# Ingestion runtime deps for the Airflow environment.
# Kept intentionally minimal (no dbt, no streamlit, no boto3) so nothing here
# clashes with Airflow's own pinned dependencies. dbt lives in its own venv,
# see docker/airflow/Dockerfile.
requests>=2.32,<3
pandas>=2.2,<3
pyarrow>=16,<21
python-dotenv>=1.2.2,<2
websocket-client>=1.8,<2
# S3Writer (used by write_bronze) talks to MinIO/S3 via boto3. The DAG imports it
# at parse time, so boto3 must live in Airflow's env. Pinned range does not clash
# with Airflow 2.10.5's transitive botocore.
boto3>=1.34,<2

# Cosmos renders the dbt project into one Airflow task per model/test. It runs
# in Airflow's env but calls dbt via SUBPROCESS against /opt/dbt-venv, so dbt's
# own deps stay isolated (astronomer-cosmos itself does NOT pull in dbt-core).
astronomer-cosmos>=1.7,<2

# Great Expectations powers the validate_bronze quality gate. It runs in
# Airflow's env (validate_bronze is a PythonOperator) but is imported lazily
# inside the task, not at DAG parse time, so its heavy deps (scipy/cryptography)
# don't slow scheduler parsing. Tested on 1.18.2 / Python 3.11.
great-expectations>=1.0,<2
